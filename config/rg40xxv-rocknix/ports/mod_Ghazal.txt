# Ports source this after control.txt:
#   [ -f "${controlfolder}/mod_${CFW_NAME}.txt" ] && source "${controlfolder}/mod_${CFW_NAME}.txt"
# CFW_NAME is Ghazal, so PortMaster's per-system files (mod_ROCKNIX.txt,
# libgl_muOS.txt ...) do not apply; ports fall back to libgl_default.txt.

# Resolution options for Godot ports.
GODOT2_OPTS="-r ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"
GODOT_OPTS="--resolution ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"

# Called by ports just before the game starts. There is no window manager or
# foreground-process registry to tell; only PortMaster's own dialog must go.
pm_platform_helper() {
    if [ -e "$PM_PIPE" ]; then
        PortMasterDialogExit
    fi
}
